cmake_minimum_required(VERSION 3.18)
project(fcitx5_android_plugin_meltype LANGUAGES C CXX)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "Meltype Android plugin requires arm64-v8a")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(MELTYPE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
find_package(fcitx5 REQUIRED CONFIG)
get_target_property(FCITX5_CMAKE_MODULES fcitx5::cmake INTERFACE_INCLUDE_DIRECTORIES)
list(PREPEND CMAKE_MODULE_PATH ${FCITX5_CMAKE_MODULES})
find_package(Fcitx5Core REQUIRED MODULE)
foreach(input MELTYPE_NATIVE_LIBRARY MOZC_SOURCE_DIR MOZC_DATA_HEADER PROTOC_EXECUTABLE MELTYPE_JSON_INCLUDE_DIR)
    if(NOT IS_ABSOLUTE "${${input}}" OR NOT EXISTS "${${input}}")
        message(FATAL_ERROR "${input} must name an existing absolute build input")
    endif()
endforeach()
if(NOT EXISTS "${MELTYPE_JSON_INCLUDE_DIR}/nlohmann/json.hpp")
    message(FATAL_ERROR "MELTYPE_JSON_INCLUDE_DIR must contain nlohmann/json.hpp")
endif()
# Do not add a host /usr/include to Android's compiler search path.
file(COPY "${MELTYPE_JSON_INCLUDE_DIR}/nlohmann" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/json-include")
set(MELTYPE_BRIDGE_SOURCE_DIR "${MELTYPE_ROOT}" CACHE PATH "Explicit bridge source tree")
set_property(GLOBAL PROPERTY JOB_POOLS meltype_compile=2)
set(CMAKE_JOB_POOL_COMPILE meltype_compile)
add_subdirectory("${MELTYPE_BRIDGE_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/mozc-bridge")
add_library(MeltypeNative SHARED IMPORTED)
set_target_properties(MeltypeNative PROPERTIES IMPORTED_LOCATION "${MELTYPE_NATIVE_LIBRARY}" IMPORTED_NO_SONAME TRUE)
add_library(meltype MODULE "${MELTYPE_ROOT}/addon/engine.cpp" "${MELTYPE_ROOT}/addon/android_factory.cpp")
target_include_directories(meltype PRIVATE "${MELTYPE_ROOT}/addon" "${CMAKE_CURRENT_BINARY_DIR}/json-include")
target_link_libraries(meltype PRIVATE Fcitx5::Core MeltypeNative meltype-mozc-bridge log)
set_target_properties(meltype PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN" BUILD_WITH_INSTALL_RPATH TRUE)
target_compile_options(meltype PRIVATE -Wall -Wextra -Werror)
# Android's CMake platform does not emit the usual INSTALL_RPATH property.
target_link_options(meltype PRIVATE "LINKER:-rpath,$ORIGIN" "LINKER:-z,max-page-size=16384" "LINKER:--exclude-libs,ALL")
# FcitxComponentPlugin builds these targets and installs their named components.
# Fcitx5Core's CMake modules already define the component helper targets.
install(FILES "${MELTYPE_ROOT}/addon/meltype.conf" DESTINATION share/fcitx5/addon COMPONENT config)
install(FILES "${MELTYPE_ROOT}/addon/meltype-inputmethod.conf" DESTINATION share/fcitx5/inputmethod RENAME meltype.conf COMPONENT config)
install(FILES "${MELTYPE_ROOT}/LICENSE" "${MELTYPE_ROOT}/NOTICE" "${MELTYPE_ROOT}/THIRD-PARTY-NOTICES.md" DESTINATION share/fcitx5/meltype COMPONENT config)
install(DIRECTORY "${MELTYPE_ROOT}/licenses/" DESTINATION share/fcitx5/meltype/licenses COMPONENT config)
install(FILES "${MOZC_SOURCE_DIR}/mozc/LICENSE" DESTINATION share/fcitx5/meltype/licenses RENAME Mozc-LICENSE COMPONENT config)
install(FILES "${MOZC_SOURCE_DIR}/protobuf/LICENSE" DESTINATION share/fcitx5/meltype/licenses RENAME protobuf-LICENSE COMPONENT config)
install(FILES "${MOZC_SOURCE_DIR}/abseil-cpp/LICENSE" DESTINATION share/fcitx5/meltype/licenses RENAME Abseil-LICENSE COMPONENT config)
install(FILES "${MOZC_SOURCE_DIR}/mozc/src/data/installer/credits_en.html" DESTINATION share/fcitx5/meltype/licenses RENAME Mozc-CREDITS.html COMPONENT config)
