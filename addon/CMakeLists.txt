cmake_minimum_required(VERSION 3.18)
project(meltype_fcitx_addon LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Fcitx5Core REQUIRED)
find_package(nlohmann_json 3.2.0 REQUIRED)
set(MELTYPE_NATIVE_LIBRARY "" CACHE FILEPATH "Absolute path to Linux libMeltypeNative.so")
if(NOT EXISTS "${MELTYPE_NATIVE_LIBRARY}")
    message(FATAL_ERROR "Set MELTYPE_NATIVE_LIBRARY to the existing NativeAOT shared library")
endif()
if(ANDROID)
    message(FATAL_ERROR "Android factory/bridge and packaging are tracked in Issue #5; this entry builds the Linux adapter")
endif()
add_library(MeltypeNative SHARED IMPORTED)
set_target_properties(MeltypeNative PROPERTIES IMPORTED_LOCATION "${MELTYPE_NATIVE_LIBRARY}")
add_library(meltype-engine STATIC engine.cpp)
set_target_properties(meltype-engine PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(meltype-engine PUBLIC Fcitx5::Core nlohmann_json::nlohmann_json MeltypeNative)
target_compile_options(meltype-engine PRIVATE -Wall -Wextra -Werror)
add_library(meltype MODULE factory.cpp)
target_link_libraries(meltype PRIVATE meltype-engine)
# Keep the NativeAOT provider explicit; never copy unlicensed binaries into this repo.
get_filename_component(MELTYPE_NATIVE_DIRECTORY "${MELTYPE_NATIVE_LIBRARY}" DIRECTORY)
set_target_properties(meltype PROPERTIES INSTALL_RPATH "${MELTYPE_NATIVE_DIRECTORY}")
install(TARGETS meltype LIBRARY DESTINATION lib/fcitx5)
install(FILES meltype.conf DESTINATION share/fcitx5/addon)
install(FILES meltype-inputmethod.conf DESTINATION share/fcitx5/inputmethod RENAME meltype.conf)
include(CTest)
if(BUILD_TESTING)
    add_executable(test-meltype integration_test.cpp)
    target_link_libraries(test-meltype PRIVATE meltype-engine)
    add_test(NAME meltype-fcitx-integration COMMAND test-meltype)
    add_executable(test-meltype-failure failure_test.cpp engine.cpp)
    target_link_libraries(test-meltype-failure PRIVATE Fcitx5::Core nlohmann_json::nlohmann_json)
    add_test(NAME meltype-fcitx-failure COMMAND test-meltype-failure)
endif()
