cmake_minimum_required(VERSION 3.22)
project(meltype_native_bridge LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(MOZC_SOURCE_DIR "" CACHE PATH "Fixed clean fcitx5-mozc checkout")
set(PROTOC_EXECUTABLE "" CACHE FILEPATH "Host protoc matching protobuf revision")
set(MOZC_DATA_HEADER "" CACHE FILEPATH "Generated OSS mozc_data.inc")
foreach(input MOZC_SOURCE_DIR PROTOC_EXECUTABLE MOZC_DATA_HEADER)
  if(NOT EXISTS "${${input}}")
    message(FATAL_ERROR "Set ${input} to an existing explicit input")
  endif()
endforeach()
set(MELTYPE_MOZC_STAGE_DIR "${CMAKE_CURRENT_BINARY_DIR}/mozc-source" CACHE PATH "Reusable verified build staging directory")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/native/stage_mozc.py"
  "${MOZC_SOURCE_DIR}" "${MELTYPE_MOZC_STAGE_DIR}" "${MOZC_DATA_HEADER}"
  COMMAND_ERROR_IS_FATAL ANY)
set(BUILD_MOZC_ADDON OFF CACHE BOOL "No upstream addon" FORCE)
set(protobuf_BUILD_PROTOC_BINARIES OFF CACHE BOOL "Host protoc is supplied" FORCE)
set(protobuf_BUILD_TESTS OFF CACHE BOOL "No protobuf tests" FORCE)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/mozc-build/base" "${CMAKE_CURRENT_BINARY_DIR}/mozc-build/dictionary" "${CMAKE_CURRENT_BINARY_DIR}/mozc-build/protocol" "${CMAKE_CURRENT_BINARY_DIR}/mozc-build/data_manager" "${CMAKE_CURRENT_BINARY_DIR}/mozc-build/prediction")
add_subdirectory("${MELTYPE_MOZC_STAGE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/mozc-build" EXCLUDE_FROM_ALL)
add_library(meltype-mozc-bridge STATIC native/mozc_bridge.cc)
target_include_directories(meltype-mozc-bridge PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/native" "${CMAKE_CURRENT_SOURCE_DIR}/native/include"
  PRIVATE "${MELTYPE_MOZC_STAGE_DIR}/mozc/src" "${MELTYPE_MOZC_STAGE_DIR}/abseil-cpp"
  "${MELTYPE_MOZC_STAGE_DIR}/protobuf/src" "${CMAKE_CURRENT_BINARY_DIR}/mozc-build")
set_target_properties(meltype-mozc-bridge PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
target_compile_options(meltype-mozc-bridge PRIVATE -funsigned-char)
target_link_libraries(meltype-mozc-bridge PUBLIC mozc-static libprotobuf absl::flags_parse absl::log_flags)
add_dependencies(meltype-mozc-bridge gen)
if(ANDROID)
  target_link_libraries(meltype-mozc-bridge PUBLIC log)
endif()
option(MELTYPE_BRIDGE_TESTS "Build native bridge tests" OFF)
if(MELTYPE_BRIDGE_TESTS AND NOT ANDROID)
  enable_testing()
  add_executable(test-meltype-bridge native/bridge_test.cc)
  target_link_libraries(test-meltype-bridge PRIVATE meltype-mozc-bridge)
  add_test(NAME meltype-mozc-bridge COMMAND test-meltype-bridge "${CMAKE_CURRENT_BINARY_DIR}/test-profile")
endif()
